## ✅ 1. SSR/SSG — SEO 友好

### SSG（静态生成）— 构建时生成 HTML

```tsx
// src/app/blog/page.tsx — 静态博客列表
// 构建时生成，适合不常变化的内容

interface Post {
  id: number;
  title: string;
  content: string;
}

// 这个函数在构建时运行，生成静态 HTML
export async function generateStaticParams() {
  const posts: Post[] = [
    { id: 1, title: 'Hello Next.js', content: '...' },
    { id: 2, title: 'Getting Started', content: '...' },
    { id: 3, title: 'API Routes', content: '...' },
  ];
  
  return posts.map((post) => ({
    id: post.id.toString(),
  }));
}

// 每个页面都有完整的 metadata
export const metadata = {
  title: 'My Blog - Latest Articles',
  description: 'Read our latest blog posts about Next.js',
  openGraph: {
    title: 'My Blog',
    type: 'website',
  },
};

export default async function BlogPage() {
  // 服务端直接获取数据，不需要 useEffect
  const res = await fetch('https://api.example.com/posts');
  const posts = await res.json();

  return (
    <main>
      <h1>📝 Blog</h1>
      {posts.map((post: Post) => (
        <article key={post.id}>
          <h2>{post.title}</h2>
          <p>{post.content}</p>
        </article>
      ))}
    </main>
  );
}
```

### SSR（服务端渲染）— 每次请求都生成

```tsx
// src/app/dashboard/page.tsx — 实时数据
// 每次访问都从服务器获取最新数据

// 关键：不加这个配置就是 SSR（默认行为）
// export const dynamic = 'force-dynamic';  // 可选：强制每次渲染

export const metadata = {
  title: 'Dashboard - Real-time Data',
};

export default async function DashboardPage() {
  // 不缓存，每次请求都获取最新数据
  const stats = await fetch('https://api.example.com/stats', {
    cache: 'no-store',  // ← 关键：禁用缓存
  });
  
  const data = await stats.json();

  return (
    <div>
      <h1>📊 Dashboard</h1>
      <p>当前在线用户: {data.onlineUsers}</p>
      <p>今日订单: {data.todayOrders}</p>
    </div>
  );
}
```

## ✅ 3. Image & Font 自动优化

### Image Optimization（自动压缩/WebP/懒加载）

```tsx
import Image from 'next/image';

export default function GalleryPage() {
  return (
    <div>
      {/* ✅ 基础用法 */}
      <Image
        src="/hero.jpg"
        alt="Hero image"
        width={800}
        height={500}
        // 自动优化：
        // - 转换为 WebP 格式（浏览器支持的话）
        // - 根据设备分辨率提供不同尺寸
        // - 默认 lazy loading
      />

      {/* ✅ 响应式图片 */}
      <div className="grid grid-cols-3">
        {[1, 2, 3, 4, 5, 6].map((i) => (
          <Image
            key={i}
            src={`/gallery/photo${i}.jpg`}
            alt={`Photo ${i}`}
            width={400}
            height={300}
            placeholder="blur"  // 加载时模糊效果
            blurDataURL="data:image/jpeg;base64,..."  // 或用 blurDataURL
          />
        ))}
      </div>

      {/* ✅ 远程图片（需要配置域名） */}
      <Image
        src="https://cdn.example.com/avatar.jpg"
        alt="Avatar"
        width={100}
        height={100}
        className="rounded-full"  // 可以直接加 className
      />
    </div>
  );
}

// next.config.ts 需要配置允许的远程图片域名
const nextConfig = {
  images: {
    remotePatterns: [
      {
        protocol: 'https',
        hostname: 'cdn.example.com',
      },
    ],
  },
};
```

### Font 自动优化

```tsx
import { Inter, Noto_Sans_SC } from 'next/font/google';

// Google Fonts 自动下载并自托管（不依赖 google CDN）
const inter = Inter({
  subsets: ['latin'],
  variable: '--font-inter',     // CSS 变量名
  display: 'swap',              // 避免 FOIT（不可见文字闪烁）
});

const notoSansSC = Noto_Sans_SC({  // 中文字体
  subsets: ['latin'],
  weight: ['400', '700'],
  variable: '--font-noto-sans-sc',
});

export default function RootLayout({
  children,
}: {
  children: React.ReactNode;
}) {
  return (
    <html lang="en" className={`${inter.variable} ${notoSansSC.variable}`}>
      <body style={{ fontFamily: 'var(--font-inter), var(--font-noto-sans-sc)' }}>
        {children}
        
        {/* 生成的 HTML 会包含预加载链接 */}
        {/*
          <link rel="preconnect" href="https://fonts.googleapis.com" />
          <link rel="preconnect" href="https://fonts.gstatic.com" crossorigin />
          → 字体文件会内联或本地托管，加载超快！
        */}
      </body>
    </html>
  );
}
```

## ✅ 4. API Routes 内置

```ts
// src/app/api/users/route.ts — RESTful API
import { NextRequest, NextResponse } from 'next/server';

// 模拟数据库（实际项目用 Prisma/Drizzle）
let users = [
  { id: 1, name: 'Alice', email: 'alice@example.com' },
  { id: 2, name: 'Bob', email: 'bob@example.com' },
];

// GET /api/users — 获取所有用户
export async function GET(request: NextRequest) {
  const searchParams = request.nextUrl.searchParams;
  const search = searchParams.get('search');

  let filteredUsers = users;
  
  if (search) {
    filteredUsers = users.filter(u => 
      u.name.toLowerCase().includes(search.toLowerCase())
    );
  }

  return NextResponse.json({
    success: true,
    data: filteredUsers,
    total: filteredUsers.length,
  });
}

// POST /api/users — 创建新用户
export async function POST(request: NextRequest) {
  try {
    const body = await request.json();
    
    if (!body.name || !body.email) {
      return NextResponse.json(
        { success: false, error: 'Name and email are required' },
        { status: 400 }
      );
    }

    const newUser = {
      id: users.length + 1,
      name: body.name,
      email: body.email,
      createdAt: new Date().toISOString(),
    };

    users.push(newUser);

    return NextResponse.json(
      { success: true, data: newUser },
      { status: 201 }
    );
  } catch (error) {
    return NextResponse.json(
      { success: false, error: 'Invalid JSON' },
      { status: 400 }
    );
  }
}
```

### 在前端调用 API：

```tsx
'use client';
import { useState } from 'react';

function UserManagement() {
  const [users, setUsers] = useState([]);

  // 获取数据
  async function fetchUsers() {
    const res = await fetch('/api/users?search=alice');
    const data = await res.json();
    setUsers(data.data);
  }

  // 创建数据
  async function createUser(name: string, email: string) {
    const res = await fetch('/api/users', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name, email }),
    });
    const data = await res.json();
    
    if (data.success) alert('创建成功！');
  }

  return (
    <div>
      <button onClick={fetchUsers}>Load Users</button>
      <ul>{users.map(u => <li key={u.id}>{u.name}</li>)}</ul>
    </div>
  );
}
```

## 📊 总结对比表

| 特性 | Vite + React | Next.js |
|------|-------------|---------|
| **SEO** | ❌ 需要额外 SSR 配置 | ✅ 开箱即用 SSG/SSR |
| **路由** | ✅ react-router 手动配 | ✅ 文件系统自动生成 |
| **图片优化** | ❌ 自己实现或装库 | ✅ `next/image` 内置 |
| **字体优化** | ❌ 手动处理 | ✅ `next/font/google` |
| **API** | ❌ 需要单独后端 | ✅ API Routes 同仓库 |
| **部署** | 🔶 需要自己配 nginx/Docker | ✅ Vercel 一键 |
| **学习曲线** | 🟢 较平缓 | 🔴 较陡峭 |
| **灵活性** | ✅ 完全自由 | 🔶 有约定限制 |

**建议学习顺序：**
1. 先掌握 **文件路由**（最常用）
2. 学 **metadata**（SEO 基础）
3. 用 **Image 组件**（性能优化）
4. 写几个 **API Route**（全栈体验）
5. 最后研究 **SSG vs SSR**（高级场景）
